function(add_peripheral_app name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE smrcore_peripherals)
    install(TARGETS ${name} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endfunction()

if(SMR_PERIPHERAL_WITH_SDK)
    add_peripheral_app(app_peripherals_bridge)
    smr_target_enable_sdk_bridge(app_peripherals_bridge)

    add_peripheral_app(app_peripherals_ft_sensor_calib)
    smr_target_enable_sdk_bridge(app_peripherals_ft_sensor_calib)

    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_peripheral_app(app_peripherals_keyboard_bridge)
        target_sources(app_peripherals_keyboard_bridge PRIVATE keyboard_input.cpp keyboard_desktop.cpp)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(KEYBOARD_XCB QUIET IMPORTED_TARGET xcb)
        endif()
        if(TARGET PkgConfig::KEYBOARD_XCB)
            target_link_libraries(app_peripherals_keyboard_bridge PRIVATE PkgConfig::KEYBOARD_XCB)
            target_compile_definitions(app_peripherals_keyboard_bridge PRIVATE SMR_KEYBOARD_HAS_XCB=1)
        else()
            message(STATUS "Keyboard: terminal steps only (install libxcb1-dev for local X11 held keys)")
        endif()
        smr_target_enable_sdk_bridge(app_peripherals_keyboard_bridge)
    endif()

endif()
